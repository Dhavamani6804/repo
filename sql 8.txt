# PowerShell script to run Exp-8 ETL on Windows MySQL
# Copy-paste this entire block into PowerShell and run.

# 1) choose mysql executable (use this if mysql is in PATH)
$mysqlExe = "mysql.exe"

# If mysql.exe is NOT in PATH, uncomment and set the full path below (adjust if necessary):
# $mysqlExe = "C:\Program Files\MySQL\MySQL Server 8.0\bin\mysql.exe"

$dir = Join-Path $env:USERPROFILE "exp8"
New-Item -Path $dir -ItemType Directory -Force | Out-Null

$sqlFile = Join-Path $dir "etl.sql"
$outFile = Join-Path $dir "results.txt"

# 2) Write the ETL SQL file
# (This is the same ETL script; it creates retail_source, retail_dw, does staging transforms and reporting.)
@'
-- EXP-8 ETL script
-- NOTE: corrected obvious typo in sample data: Age 275 -> 27 for Transaction_ID = 1

-- 1) Source DB and table
DROP DATABASE IF EXISTS retail_source;
CREATE DATABASE retail_source;
USE retail_source;

CREATE TABLE sales_transaction (
    Transaction_ID INT PRIMARY KEY,
    Date DATE,
    Customer_ID INT,
    Gender VARCHAR(10),
    Age INT,
    Product_Category VARCHAR(50),
    Quantity INT,
    Price_per_Unit DECIMAL(10,2),
    Total_Amount DECIMAL(10,2)
);

(17, '2025-01-10', 117, 'Male',   45, 'Electronics', 1,1000.00, 1000.00),
(18, '2025-01-10', 118, 'Female', 29, 'Furniture',   1, 900.00,  900.00),

-- 2) Create Data Warehouse DB and dimension/fact tables
DROP DATABASE IF EXISTS retail_dw;
CREATE DATABASE retail_dw;
USE retail_dw;

CREATE TABLE dim_customer (
    Customer_ID INT PRIMARY KEY,
    Gender VARCHAR(10),
    Age INT
);

CREATE TABLE dim_product (
    Product_ID INT AUTO_INCREMENT PRIMARY KEY,
    Product_Category VARCHAR(50)
);

CREATE TABLE dim_time (
    Date DATE PRIMARY KEY,
    Year INT,
    Month INT,
    Day INT
);

CREATE TABLE fact_sales (
    Transaction_ID INT PRIMARY KEY,
    Date DATE,
    Customer_ID INT,
    Product_ID INT,
    Quantity INT,
    Total_Amount DECIMAL(10,2),
    FOREIGN KEY (Customer_ID) REFERENCES dim_customer(Customer_ID),
    FOREIGN KEY (Product_ID) REFERENCES dim_product(Product_ID),
    FOREIGN KEY (Date) REFERENCES dim_time(Date)
);

-- 3) Extract to staging (in DW for convenience)
DROP TABLE IF EXISTS staging_sales;
CREATE TABLE staging_sales AS SELECT * FROM retail_source.sales_transaction;

-- 4) Transformations
-- Remove duplicate Transaction_IDs (keep lowest Transaction_ID row)
DELETE s1 FROM staging_sales s1
JOIN staging_sales s2
  ON s1.Transaction_ID > s2.Transaction_ID
  AND s1.Transaction_ID = s2.Transaction_ID;

-- Remove rows with NULL Quantity or Total_Amount
DELETE FROM staging_sales WHERE Total_Amount IS NULL OR Quantity IS NULL;

-- Recalculate Total_Amount where inconsistent
UPDATE staging_sales
SET Total_Amount = Quantity * Price_per_Unit
WHERE Total_Amount <> Quantity * Price_per_Unit;

-- 5) Load dimensions
-- Customer dimension
TRUNCATE TABLE dim_customer;
INSERT INTO dim_customer (Customer_ID, Gender, Age)
SELECT DISTINCT Customer_ID, Gender, Age FROM staging_sales;

-- Product dimension
TRUNCATE TABLE dim_product;
INSERT INTO dim_product (Product_Category)
SELECT DISTINCT Product_Category FROM staging_sales;

-- Time dimension
TRUNCATE TABLE dim_time;
INSERT INTO dim_time (Date, Year, Month, Day)
SELECT DISTINCT Date, YEAR(Date), MONTH(Date), DAY(Date) FROM staging_sales;

-- 6) Load fact table
TRUNCATE TABLE fact_sales;
INSERT INTO fact_sales (Transaction_ID, Date, Customer_ID, Product_ID, Quantity, Total_Amount)
SELECT s.Transaction_ID, s.Date, s.Customer_ID, p.Product_ID, s.Quantity, s.Total_Amount
FROM staging_sales s
JOIN dim_product p ON s.Product_Category = p.Product_Category;

-- 7) Data quality checks and reporting queries
-- Write outputs (counts, null checks, and reports)
SELECT 'FACT_COUNT' AS metric, COUNT(*) AS value FROM fact_sales;
SELECT 'CUSTOMER_COUNT' AS metric, COUNT(DISTINCT Customer_ID) AS value FROM dim_customer;
SELECT 'FACT_NULL_TOTAL' AS metric, COUNT(*) AS value FROM fact_sales WHERE Total_Amount IS NULL;

-- Reporting 1: Total Sales by Product Category
SELECT 'REPORT: Total Sales by Product Category' AS report;
SELECT p.Product_Category, SUM(f.Total_Amount) AS Total_Sales
FROM fact_sales f JOIN dim_product p ON f.Product_ID = p.Product_ID
GROUP BY p.Product_Category ORDER BY Total_Sales DESC;

-- Reporting 2: Average Spending by Gender
SELECT 'REPORT: Average Spending by Gender' AS report;
SELECT c.Gender, ROUND(AVG(f.Total_Amount),2) AS Avg_Spending
FROM fact_sales f JOIN dim_customer c ON f.Customer_ID = c.Customer_ID
GROUP BY c.Gender;

-- Reporting 3: Monthly Sales Trend
SELECT 'REPORT: Monthly Sales' AS report;
SELECT t.Month, SUM(f.Total_Amount) AS Monthly_Sales
FROM fact_sales f JOIN dim_time t ON f.Date = t.Date
GROUP BY t.Month ORDER BY t.Month;
'@ | Out-File -FilePath $sqlFile -Encoding UTF8

# 3) Execute the SQL file and capture output
Write-Host "Running ETL SQL..." -ForegroundColor Cyan
# Run mysql and tee output to results.txt. This will prompt for password if you use -p.
# If your root has no password you can remove -p. Using -p will prompt you to type password.
& $mysqlExe -u root -p < $sqlFile 2>&1 | Tee-Object -FilePath $outFile

Write-Host "Done. Results saved to: $outFile" -ForegroundColor Green
Write-Host "Open the file or show it to your mentor:"
Write-Host $outFile
